13 Status validate_header_fields(const FileHeaderSection& header)
14 {
15 if (header.magic != FILE_HEADER_MAGIC) {
16 return Status{
17 StatusCode::BadMagic,
18 std::format(
19 "invalid SSTable header magic: expected=0x{:08x}, actual=0x{:08x}",
20 FILE_HEADER_MAGIC,
21 header.magic
22 )
23 };
24 }
25
26 // Version 0 has never been a valid SSTable version. Versions newer than
27 // this reader understands must also be rejected.
28 if (header.version == 0 || header.version > LATEST_SSTABLE_VERSION) {
29 return Status{
30 StatusCode::UnsupportedVersion,
31 std::format(
32 "unsupported SSTable version: file_version={}, latest_supported_version={}",
33 header.version,
34 LATEST_SSTABLE_VERSION
35 )
36 };
37 }
38
39 if (header.block_size != BLOCK_SIZE) {
40 return Status{
41 StatusCode::UnsupportedBlockSize,
42 std::format(
43 "unsupported SSTable block size: file_block_size={}, expected_block_size={}",
44 header.block_size,
45 BLOCK_SIZE
46 )
47 };
48 }
49
50 return Status::ok();
51 }

54Status FileHeaderSection::write(WritableFile& file, std::uint64_t& offset)
55{
56 // Reject an invalid in-memory header before adding padding or writing bytes.
57 Status status = validate_header_fields(*this);
58 if (!status.is_ok()) {
59 return status;
60 }
61
62 status = calculate_crc32(crc32);
63 if (!status.is_ok()) {
64 return status;
65 }
66
67 status = align_to_block_boundary(file, offset, BLOCK_SIZE);
68 if (!status.is_ok()) {
69 return status;
70 }
71
72 const std::uint64_t header_offset = offset;
73
74 const auto write_u32 = [&](std::uint32_t value) -> Status {
75 return kvdb::blockio::write_u32_t_le(file, value, offset, BLOCK_SIZE);
76 };
77
78 for (const std::uint32_t value : {
79 magic,
80 version,
81 flags,
82 block_size,
83 table_id,
84 crc32
85 }) {
86 status = write_u32(value);
87 if (!status.is_ok()) {
88 // Do not reset offset here. A failed append may already have written
89 // bytes, so offset must continue to describe the physical cursor.
90 return status;
91 }
92 }
93
94 Result<std::uint64_t> position = file.current_position();
95 if (!position.is_ok()) {
96 return std::move(position.status);
97 }
98
99 if (position.value != offset || offset - header_offset != disk_size()) {
100 return Status{
101 StatusCode::InvariantViolation,
102 "SSTable header write ended at an unexpected file position"
103 };
104 }
105
106 return Status::ok();
107}

109Result<FileHeaderSection> FileHeaderSection::load(
110 ReadableFile& file,
111 std::uint64_t& offset
112)
113{
114 const std::uint64_t initial_offset = offset;
115
116
117 const auto fail = [&](Status status) -> Result<FileHeaderSection> {
118 offset = initial_offset;
119 return Result<FileHeaderSection>::fail(std::move(status));
120 };
121
122 Status status = align_to_block_boundary(file, offset, BLOCK_SIZE);
123 if (!status.is_ok()) {
124 return fail(std::move(status));
125 }
126
127 const std::uint64_t header_offset = offset;
128 FileHeaderSection result{};
129
130 const auto read_u32 = [&](std::uint32_t& value) -> Status {
131 return kvdb::blockio::read_u32_t_le(file, value, offset, BLOCK_SIZE);
132 };
133
134 for (std::uint32_t* field : {
135 &result.magic,
136 &result.version,
137 &result.flags,
138 &result.block_size,
139 &result.table_id,
140 &result.crc32
141 }) {
142 status = read_u32(*field);
143 if (!status.is_ok()) {
144 return fail(std::move(status));
145 }
146 }
147
148 if (offset - header_offset != disk_size()) {
149 return fail(Status{
150 StatusCode::InvariantViolation,
151 "SSTable header load consumed an unexpected number of bytes"
152 });
153 }
154
155 if (result.magic != FILE_HEADER_MAGIC) {
156 return fail(Status{
157 StatusCode::BadMagic,
158 std::format(
159 "invalid SSTable header magic: expected=0x{:08x}, actual=0x{:08x}",
160 FILE_HEADER_MAGIC,
161 result.magic
162 )
163 });
164 }
165
166 std::uint32_t expected_crc32 = 0;
167 status = result.calculate_crc32(expected_crc32);
168 if (!status.is_ok()) {
169 return fail(std::move(status));
170 }
171
172 // Verify integrity before interpreting mutable fields such as version and
173 // block size. A flipped version bit is corruption, not a genuine new format.
174 if (expected_crc32 != result.crc32) {
175 return fail(Status{
176 StatusCode::ChecksumMismatch,
177 std::format(
178 "SSTable header CRC mismatch: expected={}, actual={}",
179 expected_crc32,
180 result.crc32
181 )
182 });
183 }
184
185 status = validate_header_fields(result);
186 if (!status.is_ok()) {
187 return fail(std::move(status));
188 }
189
190 return Result<FileHeaderSection>::ok(std::move(result));
191}
