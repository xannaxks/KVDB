27// Finalization mix - force all bits of a hash block to avalanche
28
29FORCE_INLINE uint32_t fmix(uint32_t h)
30{
31 h ^= h >> 16;
32 h *= 0x85ebca6b;
33 h ^= h >> 13;
34 h *= 0xc2b2ae35;
35 h ^= h >> 16;
36
37 return h;
38}

55void MurmurHash3_x86_32(const void* key, int len,
56 uint32_t seed, void* out)
57{
58 const uint8_t* data = (const uint8_t*)key;
59 const int nblocks = len / 4;
60
61 uint32_t h1 = seed;
62
63 uint32_t c1 = 0xcc9e2d51;
64 uint32_t c2 = 0x1b873593;
65
66 //----------
67 // body
68
69 const uint32_t* blocks = (const uint32_t*)(data + nblocks * 4);
70
71 for (int i = -nblocks; i; i++)
72 {
73 uint32_t k1 = getblock(blocks, i);
74
75 k1 *= c1;
76 k1 = ROTL32(k1, 15);
77 k1 *= c2;
78
79 h1 ^= k1;
80 h1 = ROTL32(h1, 13);
81 h1 = h1 * 5 + 0xe6546b64;
82 }
83
84 //----------
85 // tail
86
87 const uint8_t* tail = (const uint8_t*)(data + nblocks * 4);
88
89 uint32_t k1 = 0;
90
91 switch (len & 3)
92 {
93 case 3: k1 ^= tail[2] << 16;
94 case 2: k1 ^= tail[1] << 8;
95 case 1: k1 ^= tail[0];
96 k1 *= c1; k1 = ROTL32(k1, 16); k1 *= c2; h1 ^= k1;
97 };
98
99 //----------
100 // finalization
101
102 h1 ^= len;
103
104 h1 = fmix(h1);
105
106 *(uint32_t*)out = h1;
107}

111void MurmurHash3_x86_128(const void* key, const int len,
112 uint32_t seed, void* out)
113{
114 const uint8_t* data = (const uint8_t*)key;
115 const int nblocks = len / 16;
116
117 uint32_t h1 = seed;
118 uint32_t h2 = seed;
119 uint32_t h3 = seed;
120 uint32_t h4 = seed;
121
122 uint32_t c1 = 0x239b961b;
123 uint32_t c2 = 0xab0e9789;
124 uint32_t c3 = 0x38b34ae5;
125 uint32_t c4 = 0xa1e38b93;
126
127 //----------
128 // body
129
130 const uint32_t* blocks = (const uint32_t*)(data + nblocks * 16);
131
132 for (int i = -nblocks; i; i++)
133 {
134 uint32_t k1 = getblock(blocks, i * 4 + 0);
135 uint32_t k2 = getblock(blocks, i * 4 + 1);
136 uint32_t k3 = getblock(blocks, i * 4 + 2);
137 uint32_t k4 = getblock(blocks, i * 4 + 3);
138
139 k1 *= c1; k1 = ROTL32(k1, 15); k1 *= c2; h1 ^= k1;
140
141 h1 = ROTL32(h1, 19); h1 += h2; h1 = h1 * 5 + 0x561ccd1b;
142
143 k2 *= c2; k2 = ROTL32(k2, 16); k2 *= c3; h2 ^= k2;
144
145 h2 = ROTL32(h2, 17); h2 += h3; h2 = h2 * 5 + 0x0bcaa747;
146
147 k3 *= c3; k3 = ROTL32(k3, 17); k3 *= c4; h3 ^= k3;
148
149 h3 = ROTL32(h3, 15); h3 += h4; h3 = h3 * 5 + 0x96cd1c35;
150
151 k4 *= c4; k4 = ROTL32(k4, 18); k4 *= c1; h4 ^= k4;
152
153 h4 = ROTL32(h4, 13); h4 += h1; h4 = h4 * 5 + 0x32ac3b17;
154 }
155
156 //----------
157 // tail
158
159 const uint8_t* tail = (const uint8_t*)(data + nblocks * 16);
160
161 uint32_t k1 = 0;
162 uint32_t k2 = 0;
163 uint32_t k3 = 0;
164 uint32_t k4 = 0;
165
166 switch (len & 15)
167 {
168 case 15: k4 ^= tail[14] << 16;
169 case 14: k4 ^= tail[13] << 8;
170 case 13: k4 ^= tail[12] << 0;
171 k4 *= c4; k4 = ROTL32(k4, 18); k4 *= c1; h4 ^= k4;
172
173 case 12: k3 ^= tail[11] << 24;
174 case 11: k3 ^= tail[10] << 16;
175 case 10: k3 ^= tail[9] << 8;
176 case 9: k3 ^= tail[8] << 0;
177 k3 *= c3; k3 = ROTL32(k3, 17); k3 *= c4; h3 ^= k3;
178
179 case 8: k2 ^= tail[7] << 24;
180 case 7: k2 ^= tail[6] << 16;
181 case 6: k2 ^= tail[5] << 8;
182 case 5: k2 ^= tail[4] << 0;
183 k2 *= c2; k2 = ROTL32(k2, 16); k2 *= c3; h2 ^= k2;
184
185 case 4: k1 ^= tail[3] << 24;
186 case 3: k1 ^= tail[2] << 16;
187 case 2: k1 ^= tail[1] << 8;
188 case 1: k1 ^= tail[0] << 0;
189 k1 *= c1; k1 = ROTL32(k1, 15); k1 *= c2; h1 ^= k1;
190 };
191
192 //----------
193 // finalization
194
195 h1 ^= len; h2 ^= len; h3 ^= len; h4 ^= len;
196
197 h1 += h2; h1 += h3; h1 += h4;
198 h2 += h1; h3 += h1; h4 += h1;
199
200 h1 = fmix(h1);
201 h2 = fmix(h2);
202 h3 = fmix(h3);
203 h4 = fmix(h4);
204
205 h1 += h2; h1 += h3; h1 += h4;
206 h2 += h1; h3 += h1; h4 += h1;
207
208 ((uint32_t*)out)[0] = h1;
209 ((uint32_t*)out)[1] = h2;
210 ((uint32_t*)out)[2] = h3;
211 ((uint32_t*)out)[3] = h4;
212}

216void MurmurHash3_x64_128(const void* key, const int len,
217 const uint32_t seed, void* out)
218{
219 const uint8_t* data = (const uint8_t*)key;
220 const int nblocks = len / 16;
221
222 uint64_t h1 = seed;
223 uint64_t h2 = seed;
224
225 uint64_t c1 = BIG_CONSTANT(0x87c37b91114253d5);
226 uint64_t c2 = BIG_CONSTANT(0x4cf5ad432745937f);
227
228 //----------
229 // body
230
231 const uint64_t* blocks = (const uint64_t*)(data);
232
233 for (int i = 0; i < nblocks; i++)
234 {
235 uint64_t k1 = getblock(blocks, i * 2 + 0);
236 uint64_t k2 = getblock(blocks, i * 2 + 1);
237
238 k1 *= c1; k1 = ROTL64(k1, 31); k1 *= c2; h1 ^= k1;
239
240 h1 = ROTL64(h1, 27); h1 += h2; h1 = h1 * 5 + 0x52dce729;
241
242 k2 *= c2; k2 = ROTL64(k2, 33); k2 *= c1; h2 ^= k2;
243
244 h2 = ROTL64(h2, 31); h2 += h1; h2 = h2 * 5 + 0x38495ab5;
245 }
246
247 //----------
248 // tail
249
250 const uint8_t* tail = (const uint8_t*)(data + nblocks * 16);
251
252 uint64_t k1 = 0;
253 uint64_t k2 = 0;
254
255 switch (len & 15)
256 {
257 case 15: k2 ^= uint64_t(tail[14]) << 48;
258 case 14: k2 ^= uint64_t(tail[13]) << 40;
259 case 13: k2 ^= uint64_t(tail[12]) << 32;
260 case 12: k2 ^= uint64_t(tail[11]) << 24;
261 case 11: k2 ^= uint64_t(tail[10]) << 16;
262 case 10: k2 ^= uint64_t(tail[9]) << 8;
263 case 9: k2 ^= uint64_t(tail[8]) << 0;
264 k2 *= c2; k2 = ROTL64(k2, 33); k2 *= c1; h2 ^= k2;
265
266 case 8: k1 ^= uint64_t(tail[7]) << 56;
267 case 7: k1 ^= uint64_t(tail[6]) << 48;
268 case 6: k1 ^= uint64_t(tail[5]) << 40;
269 case 5: k1 ^= uint64_t(tail[4]) << 32;
270 case 4: k1 ^= uint64_t(tail[3]) << 24;
271 case 3: k1 ^= uint64_t(tail[2]) << 16;
272 case 2: k1 ^= uint64_t(tail[1]) << 8;
273 case 1: k1 ^= uint64_t(tail[0]) << 0;
274 k1 *= c1; k1 = ROTL64(k1, 29); k1 *= c2; h1 ^= k1;
275 };
276
277 //----------
278 // finalization
279
280 h1 ^= len; h2 ^= len;
281
282 h1 += h2;
283 h2 += h1;
284
285 h1 = fmix(h1);
286 h2 = fmix(h2);
287
288 h1 += h2;
289 h2 += h1;
290
291 ((uint64_t*)out)[0] = h1;
292 ((uint64_t*)out)[1] = h2;
293}
