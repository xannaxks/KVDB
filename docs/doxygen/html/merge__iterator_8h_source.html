16/** @brief Internal-key ordering: user key ascending, sequence descending. */
17static bool internal_before(const InternalRecord& a, const InternalRecord& b)
18{
19 if (a.key_entry < b.key_entry) return true;
20 if (b.key_entry < a.key_entry) return false;
21
22 // Same user key: newer sequence comes first.
23 return a.seq_num > b.seq_num;
24}
