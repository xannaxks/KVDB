23 ) noexcept
24 {

50 Result<std::optional<CompactionPlan>> pick_level_compaction(
51 const LevelManager& manager,
52 std::uint32_t level,
53 const CompactionOptions& options
54 )
55 {
56 const auto* source = tables_at(manager, level);
57 if (source == nullptr || source->empty()) {
58 return Result<std::optional<CompactionPlan>>::fail(
59 Status{
60 StatusCode::InvalidState,
61 "Compaction source level is missing or empty"
62 }
63 );
64 }
65
66 if (level + 1 >= options.max_levels) {
67 return Result<std::optional<CompactionPlan>>::fail(
68 Status{
69 StatusCode::InvalidState,
70 "Cannot compact the bottommost level"
71 }
72 );
73 }
74
75 CompactionPlan plan;
76 plan.source_level = level;
77 plan.target_level = level + 1;
78 plan.max_output_file_size =
79 options.target_file_size_per_level[plan.target_level];
80
81 if (level == 0) {
82 plan.reason = CompactionReason::L0ReachedLimit;
83 plan.source_tables = *source;
84 }
85 else {
86 plan.reason = CompactionReason::LxReachedLimit;
87
88 // Without a persisted compaction pointer, selecting the largest table
89 // is a better default than always selecting the rightmost key range.
90 const auto selected = std::max_element(
91 source->begin(),
92 source->end(),
93 [](const TableMeta& lhs, const TableMeta& rhs) {
94 if (lhs.file_size != rhs.file_size) {
95 return lhs.file_size < rhs.file_size;
96 }
97 return lhs.table_id > rhs.table_id;
98 }
99 );
100 plan.source_tables.push_back(*selected);
101 }
102
103 plan.smallest_key = plan.source_tables.front().smallest_key;
104 plan.largest_key = plan.source_tables.front().largest_key;
105
106 for (const auto& table : plan.source_tables) {
107 widen_plan_range(plan, table);
108 }
109
110 plan.overlapping_tables = manager.find_overlapping_tables(
111 plan.target_level,
112 plan.smallest_key,
113 plan.largest_key
114 );
115
116 for (const auto& table : plan.overlapping_tables) {
117 widen_plan_range(plan, table);
118 }
119
120 if (!plan.validate()) {
121 return Result<std::optional<CompactionPlan>>::fail(
122 Status{
123 StatusCode::Corruption,
124 "Scheduler produced an invalid compaction plan"
125 }
126 );
127 }
128
129 return Result<std::optional<CompactionPlan>>::ok(std::move(plan));
130 }

143 if (l0 != nullptr && l0->size() >= options.l0_file_count_trigger) {
144 return true;
145 }
146
147 for (std::uint32_t level = 1; level + 1 < options.max_levels; ++level) {
148 if (level_size_bytes(manager, level) > options.max_bytes_per_level[level]) {
149 return true;
150 }
151 }

157 const LevelManager& manager,
158 const CompactionOptions& options
159) const

161 Status options_status = options.validate();

163 return Result<std::optional<CompactionPlan>>::fail(

165 );
166 }
167
168 const auto* l0 = tables_at(manager, 0);
169 if (l0 != nullptr && l0->size() >= options.l0_file_count_trigger) {
170 return pick_level_compaction(manager, 0, options);
171 }
172
173 for (std::uint32_t level = 1; level + 1 < options.max_levels; ++level) {
174 if (level_size_bytes(manager, level) > options.max_bytes_per_level[level]) {
175 return pick_level_compaction(manager, level, options);
176 }
177 }
178
179 return Result<std::optional<CompactionPlan>>::ok(std::nullopt);
