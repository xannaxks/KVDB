46/*
47 * Adds an unsigned integer to the CRC in canonical little-endian encoding.
48 *
49 * For example, 0x12345678 is always hashed as:
50 * 78 56 34 12
51 *
52 * regardless of host endianness.
53 */
54template <std::unsigned_integral T>
55 requires (sizeof(T) <= sizeof(std::uint64_t))
56inline void crc32_add_pod(
57 std::uint32_t& crc,
58 T value
59) noexcept
60{
61 std::array<std::byte, sizeof(T)> encoded{};
62
63 const auto wide_value = static_cast<std::uint64_t>(value);
64
65 for (std::size_t i = 0; i < encoded.size(); ++i)
66 {
67 encoded[i] = static_cast<std::byte>(
68 (wide_value >> (i * 8u)) & 0xFFu
69 );
70 }
71
72 compute_crc32(crc, encoded.data(), encoded.size());
73}
