1// A compaction first revalidates every planned input against the current level
2// snapshot. It then merges iterators by internal-key order, retains the newest
3// required version per user key, rolls output builders at the target size, and
4// returns one VersionEdit that atomically adds outputs and removes all inputs.

21 bool level_contains_table(
22 const LevelManager& manager,
23 std::uint32_t level,
24 std::uint64_t table_id
25 ) noexcept
26 {
27 const auto* tables = manager.get_lx_tables(level);
28 if (tables == nullptr) {
29 return false;
30 }
31
32 return std::any_of(
33 tables->begin(),
34 tables->end(),
35 [table_id](const TableMeta& table) {
36 return table.table_id == table_id;
37 }
38 );
39 }

41 bool plan_is_current(
42 const CompactionPlan& plan,
43 const LevelManager& manager
44 ) noexcept
45 {
46 for (const auto& table : plan.source_tables) {
47 if (!level_contains_table(manager, plan.source_level, table.table_id)) {
48 return false;
49 }
50 }
51
52 for (const auto& table : plan.overlapping_tables) {
53 if (!level_contains_table(manager, plan.target_level, table.table_id)) {
54 return false;
55 }
56 }
57
58 return true;
59 }
60
61 bool is_bottommost_destination(
62 const LevelManager& manager,
63 std::uint32_t target_level
64 ) noexcept
65 {
66 if (target_level == std::numeric_limits<std::uint32_t>::max()) {
67 return true;
68 }
69
70 for (std::uint32_t level = target_level + 1; ; ++level) {
71 const auto* tables = manager.get_lx_tables(level);
72 if (tables == nullptr) {
73 return true;
74 }
75
76 if (!tables->empty()) {
77 return false;
78 }
79
80 if (level == std::numeric_limits<std::uint32_t>::max()) {
81 return true;
82 }
83 }
84 }

87Result<std::optional<VersionEdit>> CompactionJob::run(
88 const CompactionPlan& plan,
89 const LevelManager& level_manager,
90 const Manifest& manifest,
91 SSTableManager& sstable_manager,
92 Arena& arena
93) const
94{
95 if (!plan.validate()) {
96 return Result<std::optional<VersionEdit>>::fail(
97 Status{
98 StatusCode::Corruption,
99 "Compaction plan is not valid"
100 }
101 );
102 }
103
104 if (!plan_is_current(plan, level_manager)) {
105 return Result<std::optional<VersionEdit>>::fail(
106 Status{
107 StatusCode::InvalidState,
108 "Compaction plan is stale; one or more input tables changed"
109 }
110 );
111 }
112
113 VersionEdit edit;
114 std::vector<std::shared_ptr<SSTable>> pinned_tables;
115 std::vector<SSTableIterator> iterators;
116
117 const std::size_t input_count =
118 plan.source_tables.size() + plan.overlapping_tables.size();
119 pinned_tables.reserve(input_count);
120 iterators.reserve(input_count);
121
122 auto cleanup_finished_outputs = [&]() noexcept {
123 for (const auto& table : edit.payload.new_tables) {
124 if (table.path.empty()) {
125 continue;
126 }
127 std::error_code ignored;
128 std::filesystem::remove(table.path, ignored);
129 }
130 edit.payload.new_tables.clear();
131 };
132
133 auto fail = [&](Status status)
134 -> Result<std::optional<VersionEdit>> {
135 cleanup_finished_outputs();
136 return Result<std::optional<VersionEdit>>::fail(std::move(status));
137 };
138
139 auto add_input_table = [&](const TableMeta& table_meta,
140 std::uint32_t level) -> Status {
141 if (table_meta.table_id > std::numeric_limits<std::uint32_t>::max()) {
142 return Status{
143 StatusCode::InvalidState,
144 "SSTableManager currently supports only 32-bit input table IDs"
145 };
146 }
147
148 Result<std::shared_ptr<SSTable>> sstable_result =
149 sstable_manager.get(
150 static_cast<std::uint32_t>(table_meta.table_id),
151 arena
152 );
153
154 if (!sstable_result.is_ok()) {
155 return std::move(sstable_result.status);
156 }
157
158 if (!sstable_result.value) {
159 return Status{
160 StatusCode::NotFound,
161 "SSTable not found during compaction"
162 };
163 }
164
165 pinned_tables.push_back(std::move(sstable_result.value));
166 SSTable& sstable = *pinned_tables.back();
167
168 Result<std::unique_ptr<ReadableFile>> file_result =
169 open_readable_file(sstable.get_final_path());
170 if (!file_result.is_ok()) {
171 return std::move(file_result.status);
172 }
173
174 iterators.emplace_back(
175 sstable,
176 std::move(file_result.value),
177 arena
178 );
179
180 // SSTableIterator starts invalid. Initialize it explicitly rather than
181 // relying on undocumented MergeIterator::build behavior.
182 Status seek_status = iterators.back().seek_to_first();
183 if (!seek_status.is_ok()) {
184 iterators.pop_back();
185 pinned_tables.pop_back();
186 return seek_status;
187 }
188
189 edit.payload.deleted_tables.push_back(DeletedTable{
190 level,
191 table_meta.table_id
192 });
193 return Status::ok();
194 };
195
196 for (const auto& table : plan.source_tables) {
197 Status status = add_input_table(table, plan.source_level);
198 if (!status.is_ok()) {
199 return fail(std::move(status));
200 }
201 }
202
203 for (const auto& table : plan.overlapping_tables) {
204 Status status = add_input_table(table, plan.target_level);
205 if (!status.is_ok()) {
206 return fail(std::move(status));
207 }
208 }
209
210 if (iterators.empty()) {
211 return Result<std::optional<VersionEdit>>::ok(std::nullopt);
212 }
213
214 MergeIterator merge_iterator;
215 merge_iterator.build(iterators);
216
217 std::unique_ptr<SSTableStreamingBuilder> builder;
218 std::uint64_t next_output_table_id = manifest.next_table_id();
219
220 auto start_new_output = [&]() -> Status {
221 if (builder) {
222 return Status{
223 StatusCode::InvalidState,
224 "Tried to start an output while another builder is active"
225 };
226 }
227
228 if (next_output_table_id > std::numeric_limits<std::uint32_t>::max()) {
229 return Status{
230 StatusCode::InvalidState,
231 "SSTableManager currently supports only 32-bit table IDs"
232 };
233 }
234
235 builder = sstable_manager.create_streaming_builder(
236 static_cast<std::uint32_t>(next_output_table_id)
237 );
238 if (!builder) {
239 return Status{
240 StatusCode::InvalidState,
241 "SSTableManager returned a null streaming builder"
242 };
243 }
244
245 ++next_output_table_id;
246 return Status::ok();
247 };
248
249 auto finish_current_output = [&]() -> Status {
250 if (!builder) {
251 return Status::ok();
252 }
253
254 if (builder->empty()) {
255 builder.reset();
256 return Status::ok();
257 }
258
259 Result<std::optional<TableMeta>> meta_result =
260 builder->finish(plan.target_level, arena);
261 if (!meta_result.is_ok()) {
262 return std::move(meta_result.status);
263 }
264
265 if (!meta_result.value.has_value()) {
266 return Status{
267 StatusCode::InvalidState,
268 "Non-empty compaction output produced no TableMeta"
269 };
270 }
271
272 edit.payload.new_tables.push_back(
273 std::move(meta_result.value.value())
274 );
275 builder.reset();
276 return Status::ok();
277 };
278
279 const bool bottommost = is_bottommost_destination(
280 level_manager,
281 plan.target_level
282 );
283
284 while (merge_iterator.valid()) {
285 const InternalRecord& newest = merge_iterator.record();
286 const ArenaEntry current_key = newest.key_entry;
287
288 if (compaction_keep_newest_record(newest.type, bottommost)) {
289 if (!builder) {
290 Status start_status = start_new_output();
291 if (!start_status.is_ok()) {
292 return fail(std::move(start_status));
293 }
294 }
295
296 Status add_status = builder->add(newest);
297 if (!add_status.is_ok()) {
298 return fail(std::move(add_status));
299 }
300 }
301
302 // No-snapshot policy: discard every older version of this user key.
303 merge_iterator.next();
304 while (merge_iterator.valid() &&
305 merge_iterator.record().key_entry == current_key) {
306 merge_iterator.next();
307 }
308
309 if (builder &&
310 !builder->empty() &&
311 builder->approximate_disk_space() >= plan.max_output_file_size) {
312 Status finish_status = finish_current_output();
313 if (!finish_status.is_ok()) {
314 return fail(std::move(finish_status));
315 }
316 }
317 }
318
319 // If MergeIterator stopped because an underlying iterator failed, do not
320 // commit a partial compaction.
321 for (const auto& iterator : iterators) {
322 Status iterator_status = iterator.status();
323 if (!iterator_status.is_ok()) {
324 return fail(std::move(iterator_status));
325 }
326 }
327
328 Status finish_status = finish_current_output();
329 if (!finish_status.is_ok()) {
330 return fail(std::move(finish_status));
331 }
332
333 if (next_output_table_id != manifest.next_table_id()) {
334 edit.payload.next_table_id = next_output_table_id;
335 }
336
337 if (edit.payload.deleted_tables.empty() &&
338 edit.payload.new_tables.empty()) {
339 return Result<std::optional<VersionEdit>>::ok(std::nullopt);
340 }
341
342 // Pick/run/Manifest::commit must be serialized. The job reserves IDs from
343 // manifest.next_table_id() and records the next value in this edit.
344 return Result<std::optional<VersionEdit>>::ok(std::move(edit));
345}
