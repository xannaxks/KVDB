72std::optional<InternalRecord> InternalRecord::read(std::ifstream& file, Arena& arena)
73{
74 const Arena::Checkpoint checkpoint = arena.checkpoint();
75 InternalRecord result;
76
77 auto seq_num_opt = kvdb::endian::read_u64_le(file);
78 if (!seq_num_opt.has_value())
79 return std::nullopt;
80
81 result.seq_num = *seq_num_opt;
82
83 auto type_opt = kvdb::endian::read_u8(file);
84 if (!type_opt.has_value())
85 return std::nullopt;
86
87 const auto type_raw = *type_opt;
88
89 if (type_raw != static_cast<std::uint8_t>(Type::Put) &&
90 type_raw != static_cast<std::uint8_t>(Type::Tombstone))
91 {
92 return std::nullopt;
93 }
94
95 result.type = static_cast<Type>(type_raw);
96
97 auto key_bytes_opt = kvdb::endian::read_bytes_with_u32_size(file);
98 if (!key_bytes_opt.has_value())
99 return std::nullopt;
100
101 auto value_bytes_opt = kvdb::endian::read_bytes_with_u32_size(file);
102 if (!value_bytes_opt.has_value())
103 return std::nullopt;
104
105 Result<ArenaEntry> key_result = ArenaEntry::make_entry(
106 arena,
107 std::span<const std::byte>(key_bytes_opt->data(), key_bytes_opt->size())
108 );
109 if (!key_result.is_ok()) {
110 arena.rollback(checkpoint);
111 return std::nullopt;
112 }
113
114 Result<ArenaEntry> value_result = ArenaEntry::make_entry(
115 arena,
116 std::span<const std::byte>(value_bytes_opt->data(), value_bytes_opt->size())
117 );
118 if (!value_result.is_ok()) {
119 arena.rollback(checkpoint);
120 return std::nullopt;
121 }
122
123 result.key_entry = key_result.value;
124 result.value_entry = value_result.value;
125
126 return std::optional<InternalRecord>{std::move(result)};
