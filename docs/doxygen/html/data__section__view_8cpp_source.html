27 [[nodiscard]] bool checked_add_u64(
28 std::uint64_t lhs,
29 std::uint64_t rhs,
30 std::uint64_t& result
31 ) noexcept
32 {
33 if (lhs > std::numeric_limits<std::uint64_t>::max() - rhs) {
34 return false;
35 }
36 result = lhs + rhs;
37 return true;
38 }
39
40 [[nodiscard]] bool checked_mul_u64(
41 std::uint64_t lhs,
42 std::uint64_t rhs,
43 std::uint64_t& result
44 ) noexcept
45 {
46 if (lhs != 0 && rhs > std::numeric_limits<std::uint64_t>::max() / lhs) {
47 return false;
48 }
49 result = lhs * rhs;
50 return true;
51 }

53 [[nodiscard]] std::uint8_t byte_value(std::byte value) noexcept
54 {
55 return std::to_integer<std::uint8_t>(value);
56 }
57
58 [[nodiscard]] bool read_u8(
59 std::span<const std::byte> bytes,
60 std::size_t& position,
61 std::uint8_t& value
62 ) noexcept
63 {
64 if (position >= bytes.size()) {
65 return false;
66 }
67 value = byte_value(bytes[position]);
68 ++position;
69 return true;
70 }
71
72 [[nodiscard]] bool read_u32_le(
73 std::span<const std::byte> bytes,
74 std::size_t& position,
75 std::uint32_t& value
76 ) noexcept
77 {
78 if (bytes.size() - position < sizeof(std::uint32_t)) {
79 return false;
80 }
81
82 value = static_cast<std::uint32_t>(byte_value(bytes[position])) |
83 (static_cast<std::uint32_t>(byte_value(bytes[position + 1])) << 8U) |
84 (static_cast<std::uint32_t>(byte_value(bytes[position + 2])) << 16U) |
85 (static_cast<std::uint32_t>(byte_value(bytes[position + 3])) << 24U);
86 position += sizeof(std::uint32_t);
87 return true;
88 }
89
90 [[nodiscard]] bool read_u64_le(
91 std::span<const std::byte> bytes,
92 std::size_t& position,
93 std::uint64_t& value
94 ) noexcept
95 {
96 if (bytes.size() - position < sizeof(std::uint64_t)) {
97 return false;
98 }
99
100 value = 0;
101 for (std::size_t i = 0; i < sizeof(std::uint64_t); ++i) {
102 value |= static_cast<std::uint64_t>(byte_value(bytes[position + i]))
103 << (i * 8U);
104 }
105 position += sizeof(std::uint64_t);
106 return true;
107 }
108
109 [[nodiscard]] bool is_valid_record_type(::Type type) noexcept
110 {
111 return type == ::Type::Put || type == ::Type::Tombstone;
112 }
113
114 [[nodiscard]] int compare_keys(
115 std::span<const std::byte> lhs,
116 std::span<const std::byte> rhs
117 ) noexcept
118 {
119 const std::size_t common = std::min(lhs.size(), rhs.size());
120 if (common > 0) {
121 const int result = std::memcmp(lhs.data(), rhs.data(), common);
122 if (result < 0) {
123 return -1;
124 }
125 if (result > 0) {
126 return 1;
127 }
128 }
129
130 if (lhs.size() < rhs.size()) {
131 return -1;
132 }
133 if (lhs.size() > rhs.size()) {
134 return 1;
135 }
136 return 0;
137 }

139 [[nodiscard]] Status corruption(std::string message)
140 {
141 return Status{ StatusCode::Corruption, std::move(message) };
142 }

150Result<DataSectionView::Header> DataSectionView::Header::load(
151 ReadableFile& file,
152 std::uint64_t& offset,
153 std::uint64_t section_end_offset
154)
155{
156 const std::uint64_t original_offset = offset;
157
158 if (offset % BLOCK_SIZE != 0) {
159 return Result<Header>::fail(Status{
160 StatusCode::InvalidAlignment,
161 "data block header offset is not block aligned"
162 });
163 }
164
165 std::uint64_t header_end = 0;
166 if (!checked_add_u64(offset, kHeaderSize, header_end) ||
167 header_end > section_end_offset) {
168 return Result<Header>::fail(Status{
169 StatusCode::UnexpectedEOF,
170 "data block header lies outside the declared data section"
171 });
172 }
173
174 Header result{};
175 result.header_offset = offset;
176
177 std::uint8_t encoded_type = 0;
178 Status status = kvdb::blockio::read_u8_t(
179 file,
180 encoded_type,
181 offset,
182 BLOCK_SIZE
183 );
184 if (!status.is_ok()) {
185 offset = original_offset;
186 return Result<Header>::fail(std::move(status));
187 }
188
189 result.header.type = static_cast<BlockType>(encoded_type);
190 status = kvdb::blockio::read_u32_t_le(
191 file,
192 result.header.payload_disk_size,
193 offset,
194 BLOCK_SIZE
195 );
196 if (!status.is_ok()) {
197 offset = original_offset;
198 return Result<Header>::fail(std::move(status));
199 }
200
201 status = kvdb::blockio::read_u32_t_le(
202 file,
203 result.header.crc32,
204 offset,
205 BLOCK_SIZE
206 );
207 if (!status.is_ok()) {
208 offset = original_offset;
209 return Result<Header>::fail(std::move(status));
210 }
211
212 if (result.header.type != BlockType::Data) {
213 offset = original_offset;
214 return Result<Header>::fail(Status{
215 StatusCode::InvalidBlockType,
216 "expected a data block while discovering DataSectionView"
217 });
218 }
219
220 constexpr std::uint32_t payload_capacity =
221 static_cast<std::uint32_t>(BLOCK_SIZE - kHeaderSize);
222 if (result.header.payload_disk_size > payload_capacity) {
223 offset = original_offset;
224 return Result<Header>::fail(Status{
225 StatusCode::InvalidPayloadSize,
226 "data block payload size exceeds physical block capacity"
227 });
228 }
229
230 result.payload_offset = offset;
231 if (!checked_add_u64(
232 result.payload_offset,
233 result.header.payload_disk_size,
234 result.payload_end_offset)) {
235 offset = original_offset;
236 return Result<Header>::fail(Status{
237 StatusCode::OffsetOutOfRange,
238 "data block payload end overflows uint64_t"
239 });
240 }
241
242 if (!checked_add_u64(
243 result.header_offset,
244 BLOCK_SIZE,
245 result.next_block_offset)) {
246 offset = original_offset;
247 return Result<Header>::fail(Status{
248 StatusCode::OffsetOutOfRange,
249 "next data block offset overflows uint64_t"
250 });
251 }
252
253 if (result.payload_end_offset > result.next_block_offset ||
254 result.next_block_offset > section_end_offset) {
255 offset = original_offset;
256 return Result<Header>::fail(Status{
257 StatusCode::OffsetOverlap,
258 "data block extends outside its physical block or declared section"
259 });
260 }
261
262 return Result<Header>::ok(std::move(result));
263}

286Result<DataSectionView::DataBlock> DataSectionView::DataBlock::load(
287 ReadableFile& file,
288 std::uint64_t& offset,
289 std::uint64_t section_end_offset
290)
291{
292 const std::uint64_t original_offset = offset;
293
294 Result<Header> header_result = Header::load(
295 file,
296 offset,
297 section_end_offset
298 );
299 if (!header_result.is_ok()) {
300 offset = original_offset;
301 return Result<DataBlock>::fail(std::move(header_result.status));
302 }
303
304 DataBlock result{};
305 result.header_view = std::move(header_result.value);
306
307 // Payload remains unread. Advance only the caller's logical discovery cursor.
308 offset = result.header_view.next_block_offset;
309 return Result<DataBlock>::ok(std::move(result));
310}

312Status DataSectionView::DataBlock::validate(ReadableFile& file) const
313{
314 if (validation_state_ == ValidationState::Valid) {
315 return Status::ok();
316 }
317 if (validation_state_ == ValidationState::Corrupt &&
318 cached_corruption_.has_value()) {
319 return *cached_corruption_;
320 }
321
322 const std::uint32_t payload_size = header_view.header.payload_disk_size;
323 constexpr std::uint32_t payload_capacity =
324 static_cast<std::uint32_t>(BLOCK_SIZE - kHeaderSize);
325
326 if (payload_size > payload_capacity) {
327 Status status = corruption(
328 "data block payload exceeds physical block capacity"
329 );
330 validation_state_ = ValidationState::Corrupt;
331 cached_corruption_ = status;
332 return status;
333 }
334
335 std::array<std::byte, BLOCK_SIZE> storage{};
336 std::uint64_t read_offset = header_view.payload_offset;
337 Status status = kvdb::blockio::read_bytes(
338 file,
339 storage.data(),
340 payload_size,
341 read_offset,
342 BLOCK_SIZE
343 );
344 if (!status.is_ok()) {
345 // Do not permanently cache transient file-system failures.
346 return status;
347 }
348
349 if (read_offset != header_view.payload_end_offset) {
350 return Status{
351 StatusCode::InvalidOffset,
352 "data payload read did not end at the declared payload boundary"
353 };
354 }
355
356 std::uint32_t actual_crc = ::crc32(0L, Z_NULL, 0);
357 if (payload_size > 0) {
358 compute_crc32(actual_crc, storage.data(), payload_size);
359 }
360
361 if (actual_crc != header_view.header.crc32) {
362 status = Status{
363 StatusCode::ChecksumMismatch,
364 "data block payload CRC does not match its header"
365 };
366 validation_state_ = ValidationState::Corrupt;
367 cached_corruption_ = status;
368 return status;
369 }
370
371 if (payload_size < kRecordFixedSize) {
372 status = corruption(
373 "data block payload is too small to contain one complete record"
374 );
375 validation_state_ = ValidationState::Corrupt;
376 cached_corruption_ = status;
377 return status;
378 }
379
380 const std::span<const std::byte> bytes(storage.data(), payload_size);
381 std::vector<RecordMeta> staged_records;
382
383 try {
384 staged_records.reserve(payload_size / kRecordFixedSize);
385 }
386 catch (const std::bad_alloc&) {
387 return Status{
388 StatusCode::BadAlloc,
389 "failed to reserve lazy data-record metadata"
390 };
391 }
392
393 std::size_t position = 0;
394 std::span<const std::byte> previous_key{};
395 std::uint64_t previous_seq = 0;
396 bool has_previous = false;
397
398 while (position < bytes.size()) {
399 const std::size_t record_start = position;
400 RecordMeta meta{};
401 std::uint8_t encoded_type = 0;
402 std::uint32_t flags = 0;
403 std::uint32_t reserved = 0;
404
405 if (bytes.size() - position < kRecordFixedSize ||
406 !read_u32_le(bytes, position, meta.key_size) ||
407 !read_u32_le(bytes, position, meta.value_size) ||
408 !read_u8(bytes, position, encoded_type) ||
409 !read_u32_le(bytes, position, flags) ||
410 !read_u32_le(bytes, position, reserved) ||
411 !read_u64_le(bytes, position, meta.seq_num)) {
412 status = corruption("truncated fixed fields in data record");
413 validation_state_ = ValidationState::Corrupt;
414 cached_corruption_ = status;
415 return status;
416 }
417
418 meta.type = static_cast<::Type>(encoded_type);
419 if (!is_valid_record_type(meta.type)) {
420 status = corruption("data record has an invalid record type");
421 validation_state_ = ValidationState::Corrupt;
422 cached_corruption_ = status;
423 return status;
424 }
425 if (flags != 0 || reserved != 0) {
426 status = corruption(
427 "data record flags and reserved fields must be zero in version 1"
428 );
429 validation_state_ = ValidationState::Corrupt;
430 cached_corruption_ = status;
431 return status;
432 }
433
434 const std::uint64_t variable_size =
435 static_cast<std::uint64_t>(meta.key_size) + meta.value_size;
436 if (variable_size > bytes.size() - position) {
437 status = corruption(
438 "data record key/value lengths exceed the remaining block payload"
439 );
440 validation_state_ = ValidationState::Corrupt;
441 cached_corruption_ = status;
442 return status;
443 }
444
445 const std::size_t key_position = position;
446 const std::size_t value_position = position + meta.key_size;
447 const std::span<const std::byte> current_key(
448 bytes.data() + key_position,
449 meta.key_size
450 );
451
452 if (has_previous) {
453 const int key_order = compare_keys(previous_key, current_key);
454 if (key_order > 0) {
455 status = corruption(
456 "data records are not ordered by ascending user key"
457 );
458 validation_state_ = ValidationState::Corrupt;
459 cached_corruption_ = status;
460 return status;
461 }
462 if (key_order == 0 && meta.seq_num >= previous_seq) {
463 status = corruption(
464 "equal data keys are not ordered by strictly descending sequence"
465 );
466 validation_state_ = ValidationState::Corrupt;
467 cached_corruption_ = status;
468 return status;
469 }
470 }
471
472 meta.record_offset = header_view.payload_offset + record_start;
473 meta.key_offset = header_view.payload_offset + key_position;
474 meta.value_offset = header_view.payload_offset + value_position;
475
476 position += static_cast<std::size_t>(variable_size);
477
478 try {
479 staged_records.emplace_back(meta);
480 }
481 catch (const std::bad_alloc&) {
482 return Status{
483 StatusCode::BadAlloc,
484 "failed to cache lazy data-record metadata"
485 };
486 }
487
488 previous_key = current_key;
489 previous_seq = meta.seq_num;
490 has_previous = true;
491 }
492
493 if (staged_records.empty() || position != bytes.size()) {
494 status = corruption(
495 "data block payload does not contain a canonical sequence of records"
496 );
497 validation_state_ = ValidationState::Corrupt;
498 cached_corruption_ = status;
499 return status;
500 }
501
502 records_ = std::move(staged_records);
503 cached_corruption_.reset();
504 validation_state_ = ValidationState::Valid;
505 return Status::ok();
506}

508Result<InternalRecord> DataSectionView::DataBlock::read_record(
509 ReadableFile& file,
510 std::size_t record_index,
511 Arena& arena
512) const
513{
514 Status status = validate(file);
515 if (!status.is_ok()) {
516 return Result<InternalRecord>::fail(std::move(status));
517 }
518
519 if (record_index >= records_.size()) {
520 return Result<InternalRecord>::fail(Status{
521 StatusCode::InvalidArgument,
522 "data record index is outside the validated block"
523 });
524 }
525
526 const RecordMeta& meta = records_[record_index];
527 const Arena::Checkpoint checkpoint = arena.checkpoint();
528
529 InternalRecord record{};
530 record.type = meta.type;
531 record.seq_num = meta.seq_num;
532 record.key_entry.size = meta.key_size;
533 record.value_entry.size = meta.value_size;
534
535 if (meta.key_size > 0) {
536 Result<void*> allocation = arena.alloc(meta.key_size, alignof(std::byte));
537 if (!allocation.is_ok()) {
538 arena.rollback(checkpoint);
539 return Result<InternalRecord>::fail(std::move(allocation.status));
540 }
541
542 record.key_entry.data = allocation.value;
543 std::uint64_t key_offset = meta.key_offset;
544 status = kvdb::blockio::read_bytes(
545 file,
546 static_cast<std::byte*>(record.key_entry.data),
547 meta.key_size,
548 key_offset,
549 BLOCK_SIZE
550 );
551 if (!status.is_ok() || key_offset != meta.value_offset) {
552 arena.rollback(checkpoint);
553 if (!status.is_ok()) {
554 return Result<InternalRecord>::fail(std::move(status));
555 }
556 return Result<InternalRecord>::fail(Status{
557 StatusCode::InvalidOffset,
558 "key read did not end at the validated value offset"
559 });
560 }
561 }
562
563 if (meta.value_size > 0) {
564 Result<void*> allocation = arena.alloc(meta.value_size, alignof(std::byte));
565 if (!allocation.is_ok()) {
566 arena.rollback(checkpoint);
567 return Result<InternalRecord>::fail(std::move(allocation.status));
568 }
569
570 record.value_entry.data = allocation.value;
571 std::uint64_t value_offset = meta.value_offset;
572 status = kvdb::blockio::read_bytes(
573 file,
574 static_cast<std::byte*>(record.value_entry.data),
575 meta.value_size,
576 value_offset,
577 BLOCK_SIZE
578 );
579 if (!status.is_ok() ||
580 value_offset != meta.record_offset + meta.disk_size()) {
581 arena.rollback(checkpoint);
582 if (!status.is_ok()) {
583 return Result<InternalRecord>::fail(std::move(status));
584 }
585 return Result<InternalRecord>::fail(Status{
586 StatusCode::InvalidOffset,
587 "value read did not end at the validated record boundary"
588 });
589 }
590 }
591
592 return Result<InternalRecord>::ok(std::move(record));
593}

595Result<std::optional<std::size_t>>
596DataSectionView::DataBlock::find_first_record(
597 ReadableFile& file,
598 const ArenaEntry& key
599) const
600{
601 if (key.size > 0 && key.data == nullptr) {
602 return Result<std::optional<std::size_t>>::fail(Status{
603 StatusCode::InvalidArgument,
604 "searched key has a non-zero size but a null data pointer"
605 });
606 }
607
608 Status status = validate(file);
609 if (!status.is_ok()) {
610 return Result<std::optional<std::size_t>>::fail(std::move(status));
611 }
612
613 const std::uint32_t payload_size = header_view.header.payload_disk_size;
614 constexpr std::uint32_t payload_capacity =
615 static_cast<std::uint32_t>(BLOCK_SIZE - kHeaderSize);
616
617 if (payload_size > payload_capacity) {
618 return Result<std::optional<std::size_t>>::fail(corruption(
619 "validated data block payload exceeds physical block capacity"
620 ));
621 }
622
623 std::array<std::byte, BLOCK_SIZE> storage{};
624 std::uint64_t read_offset = header_view.payload_offset;
625 status = kvdb::blockio::read_bytes(
626 file,
627 storage.data(),
628 payload_size,
629 read_offset,
630 BLOCK_SIZE
631 );
632 if (!status.is_ok()) {
633 return Result<std::optional<std::size_t>>::fail(std::move(status));
634 }
635
636 if (read_offset != header_view.payload_end_offset) {
637 return Result<std::optional<std::size_t>>::fail(Status{
638 StatusCode::InvalidOffset,
639 "data payload search read ended at an unexpected offset"
640 });
641 }
642
643 const std::span<const std::byte> searched_key(
644 static_cast<const std::byte*>(key.data),
645 key.size
646 );
647
648 auto record_key = [&](const RecordMeta& meta)
649 -> Result<std::span<const std::byte>>
650 {
651 if (meta.key_offset < header_view.payload_offset) {
652 return Result<std::span<const std::byte>>::fail(corruption(
653 "record key offset precedes its block payload"
654 ));
655 }
656
657 const std::uint64_t relative_u64 =
658 meta.key_offset - header_view.payload_offset;
659
660 if (relative_u64 > payload_size ||
661 meta.key_size > payload_size - relative_u64) {
662 return Result<std::span<const std::byte>>::fail(corruption(
663 "record key range lies outside its validated block payload"
664 ));
665 }
666
667 const std::size_t relative =
668 static_cast<std::size_t>(relative_u64);
669
670 return Result<std::span<const std::byte>>::ok(
671 std::span<const std::byte>(
672 storage.data() + relative,
673 meta.key_size
674 )
675 );
676 };
677
678 std::size_t first = 0;
679 std::size_t last = records_.size();
680
681 while (first < last) {
682 const std::size_t middle = first + (last - first) / 2;
683 Result<std::span<const std::byte>> current =
684 record_key(records_[middle]);
685
686 if (!current.is_ok()) {
687 return Result<std::optional<std::size_t>>::fail(
688 std::move(current.status)
689 );
690 }
691
692 if (compare_keys(current.value, searched_key) < 0) {
693 first = middle + 1;
694 }
695 else {
696 last = middle;
697 }
698 }
699
700 if (first == records_.size()) {
701 return Result<std::optional<std::size_t>>::ok(std::nullopt);
702 }
703
704 Result<std::span<const std::byte>> candidate =
705 record_key(records_[first]);
706 if (!candidate.is_ok()) {
707 return Result<std::optional<std::size_t>>::fail(
708 std::move(candidate.status)
709 );
710 }
711
712 if (compare_keys(candidate.value, searched_key) != 0) {
713 return Result<std::optional<std::size_t>>::ok(std::nullopt);
714 }
715
716 // Equal user keys are validated in strictly descending sequence order,
717 // so lower_bound returns the newest record for this key.
718 return Result<std::optional<std::size_t>>::ok(first);
719}

721Result<DataSectionView> DataSectionView::load(
722 ReadableFile& file,
723 std::uint64_t& offset,
724 std::uint64_t first_block_offset,
725 std::uint64_t data_block_count,
726 std::uint64_t expected_section_end_offset
727)
728{
729 const std::uint64_t original_offset = offset;
730
731 std::uint64_t file_size = 0;
732 Status status = file.get_file_size(file_size);
733 if (!status.is_ok()) {
734 return Result<DataSectionView>::fail(std::move(status));
735 }
736
737 if (first_block_offset > file_size) {
738 return Result<DataSectionView>::fail(Status{
739 StatusCode::OffsetOutOfRange,
740 "first data block offset lies beyond end of file"
741 });
742 }
743
744 if (data_block_count > 0 && first_block_offset % BLOCK_SIZE != 0) {
745 return Result<DataSectionView>::fail(Status{
746 StatusCode::InvalidAlignment,
747 "non-empty data section must start at a block boundary"
748 });
749 }
750
751 std::uint64_t reserved_bytes = 0;
752 std::uint64_t computed_section_end = 0;
753 if (!checked_mul_u64(data_block_count, BLOCK_SIZE, reserved_bytes) ||
754 !checked_add_u64(
755 first_block_offset,
756 reserved_bytes,
757 computed_section_end)) {
758 return Result<DataSectionView>::fail(Status{
759 StatusCode::InvalidSectionSize,
760 "data block count overflows the data section extent"
761 });
762 }
763
764 if (computed_section_end > file_size) {
765 return Result<DataSectionView>::fail(Status{
766 StatusCode::InvalidSectionSize,
767 "declared data blocks are not fully present in the file"
768 });
769 }
770
771 if (expected_section_end_offset != 0) {
772 if (expected_section_end_offset > file_size ||
773 expected_section_end_offset % BLOCK_SIZE != 0) {
774 return Result<DataSectionView>::fail(Status{
775 StatusCode::InvalidSectionSize,
776 "trusted data section end is invalid or outside the file"
777 });
778 }
779
780 if (data_block_count > 0 &&
781 expected_section_end_offset != computed_section_end) {
782 return Result<DataSectionView>::fail(Status{
783 StatusCode::InvalidSectionSize,
784 "data block count does not end at the trusted next-section offset"
785 });
786 }
787
788 if (data_block_count == 0 &&
789 first_block_offset > expected_section_end_offset) {
790 return Result<DataSectionView>::fail(Status{
791 StatusCode::InvalidSectionSize,
792 "empty data section offset lies after the next section"
793 });
794 }
795 }
796
797 if (data_block_count >
798 static_cast<std::uint64_t>(std::numeric_limits<std::size_t>::max())) {
799 return Result<DataSectionView>::fail(Status{
800 StatusCode::InvalidSectionSize,
801 "data block count cannot be represented by size_t"
802 });
803 }
804
805 DataSectionView result{};
806 result.first_data_block_offset = first_block_offset;
807 result.section_end_offset = data_block_count == 0 &&
808 expected_section_end_offset != 0
809 ? expected_section_end_offset
810 : computed_section_end;
811
812 try {
813 result.data_blocks.reserve(static_cast<std::size_t>(data_block_count));
814 }
815 catch (const std::bad_alloc&) {
816 return Result<DataSectionView>::fail(Status{
817 StatusCode::BadAlloc,
818 "failed to reserve lazy data-block metadata"
819 });
820 }
821
822 std::uint64_t cursor = first_block_offset;
823 for (std::uint64_t i = 0; i < data_block_count; ++i) {
824 Result<DataBlock> block = DataBlock::load(
825 file,
826 cursor,
827 computed_section_end
828 );
829 if (!block.is_ok()) {
830 offset = original_offset;
831 return Result<DataSectionView>::fail(std::move(block.status));
832 }
833
834 try {
835 result.data_blocks.emplace_back(std::move(block.value));
836 }
837 catch (const std::bad_alloc&) {
838 offset = original_offset;
839 return Result<DataSectionView>::fail(Status{
840 StatusCode::BadAlloc,
841 "failed to append lazy data-block metadata"
842 });
843 }
844 }
845
846 if (cursor != computed_section_end) {
847 offset = original_offset;
848 return Result<DataSectionView>::fail(Status{
849 StatusCode::InvalidSectionSize,
850 "lazy data-block discovery ended at an unexpected offset"
851 });
852 }
853
854 offset = data_block_count == 0 ? first_block_offset : computed_section_end;
855 return Result<DataSectionView>::ok(std::move(result));
856}
