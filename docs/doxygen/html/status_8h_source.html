91/**
92* @brief Indicates status of operations final result.
93*
94* Consider using with void functions.
95* In case of functions that return value, consider using Result<T>.
96*/
97struct Status {
98 Status() = default;
99 Status(const Status&) = default;
100 Status(Status&&) noexcept = default;
101
102 Status(StatusCode code, std::string message = "")
103 : code(code), message(std::move(message)) {
104 }
105
106 Status(StatusCode code, const char* message)
107 : code(code), message(message) {
108 }
109
110 StatusCode code = StatusCode::Ok; ///< default status code indicates success.
111 std::string message; ///< message providing detailed information.
112
113 /**
114 * @brief Returns success statuscode and empty message.
115 *
116 * @return Status{StatusCode::Ok}
117 */
118 static Status ok();
119
120 /**
121 * @brief Returns status indicating invalid argument with message providing detailed information.
122 *
123 * @param[in] message message with detailed information.
124 *
125 * @return Status{StatusCode::InvalidArgument, std::string}
126 */
127 static Status invalid_argument(std::string message)
128 {
129 return Status{ StatusCode::InvalidArgument, std::move(message) };
130 }
131
132 /**
133 * @brief Checks whether StatusCode is StatusCode::Ok.
134 *
135 * @retval true in case if StatusCode == StatusCode::Ok
136 * @retval false otherwise
137 */
138 bool is_ok() const;
139
140 Status& operator=(const Status&) = default;
141 Status& operator=(Status&&) noexcept = default;
142};

144/**
145* @brief Returns status of functions final result, with result itself.
146*
147* Consider using with functions that have return value. T indicates the
148* data type of return value.
149* Ex: Result<bool> test()
150* return Result<bool>::ok(true);
151*
152* @note Consider using std::optional<T> to avoid construction of empty object.
153*/
154template <typename T>
155struct [[nodiscard]] Result
156{
157 T value; ///< data type of functions return value.
158 Status status; ///< status indicating status code and message.
159
160 static Result ok(T v)
161 {
162 return Result{ std::move(v), Status::ok() };
163 }
164
165 /**
166 * @brief Use this in case of failure, return empty object
167 *
168 * @param[in] s Status object, indicating final result of function execution.
169 *
170 * @note This statis method constructs empty object, which might be expensive.
171 * @note Consider using std::optional<T> to avoid construction of empty object.
172 */
173 static Result fail(Status s)
174 {
175 return Result{ T{}, std::move(s) };
176 }
177
178 bool is_ok() const
179 {
180 return this->status.is_ok();
181 }
182};
