72Result<std::shared_ptr<SSTable>> SSTableManager::open_impl(
73 std::uint32_t table_id,
74 const std::filesystem::path& path,
75 Arena& arena
76)
77{
78 const auto cached = pool.find(table_id);
79 if (cached != pool.end()) {
80 return Result<std::shared_ptr<SSTable>>::ok(cached->second);
81 }
82
83 auto sstable_result = SSTableLoader::load(path, arena);
84 if (!sstable_result.is_ok()) {
85 return Result<std::shared_ptr<SSTable>>::fail(
86 std::move(sstable_result.status)
87 );
88 }
89
90 auto sstable_ptr = std::make_shared<SSTable>(
91 std::move(sstable_result.value)
92 );
93
94 const auto [it, inserted] = pool.emplace(table_id, sstable_ptr);
95 if (!inserted) {
96 return Result<std::shared_ptr<SSTable>>::ok(it->second);
97 }
98
99 return Result<std::shared_ptr<SSTable>>::ok(std::move(sstable_ptr));
100}
